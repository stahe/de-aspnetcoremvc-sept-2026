    4. Die View und ihr Modell

    Bisher haben die Actions ihre Antworten selbst erzeugt. In einer MVC-Anwendung ist das die Aufgabe einer View: einer Datei, die die HTML-Seite beschreibt, mit Platzhaltern für die Daten, die ihr die Action übergibt – das Modell der View. [ASP.NET Core MVC] verwendet die [Razor]-View-Engine: Eine View ist eine [.cshtml]-Datei, abgeleitet von HTML, in die [C#]-Code eingefügt wird, eingeleitet durch das Zeichen [@]. [Razor] ist Teil des Frameworks: Es muss nichts installiert werden.

using Microsoft.AspNetCore.Mvc;

namespace Example13.Controllers;

public class ViewsController : Controller
{
    // View(model): die View, die den Namen der Action trägt (Views/Views/Index.cshtml)
    [HttpGet("/")]
    public IActionResult Index()
    {
        var model = new PageModel(
            "My first view",
            "This page was built by Razor, on the server.",
            DateTime.Now.ToString("HH:mm:ss"));
        return View(model);
    }

    // View(name, model): die View wird von der Action gewählt
    [HttpGet("hello/{name}")]
    public IActionResult Hello(string name)
    {
        if (name == "unknown")
        {
            Response.StatusCode = 404;
            return View("Unknown", name);
        }
        var model = new PageModel(
            $"Hello {name}",
            "This time, the action designated the view by its name.",
            DateTime.Now.ToString("HH:mm:ss"));
        return View("Index", model);
    }
}

    • Zeilen 2–5: Ein Codeblock mit dem Namen [@{ ... }] enthält [C#]-Anweisungen. Er erzeugt nichts auf der Seite. Die darin deklarierten Variablen ([total], [count]) können später in der View verwendet werden;
    • Zeile 16: Ein [null] -Wert ([@Model.Nickname]) zeigt nichts an, ohne Fehler;
    • Zeilen 18–19: Ein Ausdruck [@...] maskiert die Sonderzeichen in HTML ([<] wird zu [&lt;] ...): Der Text wird unverändert angezeigt. [@Html.Raw(...)] fügt ihn ohne Maskierung ein: Er wird als HTML interpretiert. [Html.Raw] wird nur für HTML verwendet, das von der Anwendung erzeugt wurde, niemals für Daten, die ein Benutzer eingegeben hat: Der Benutzer könnte sonst HTML oder sogar [JavaScript] in die Seite einschleusen (ein Angriff, der als [XSS] bekannt ist, [cross-site scripting]);
    • Zeilen 21–27: eine Bedingung. Nach [@if (...)] ist der Inhalt innerhalb der geschweiften Klammern HTML: [Razor] erkennt den Beginn eines Tags und wechselt in den HTML-Modus zurück. Zu [C#] kehren wir über [@] zurück;
    • Zeilen 32–40: eine [for]-Schleife in [C#]. [var product = ...] ist eine [C#]-Anweisung, [<li>] ist HTML: [Razor] wechselt ohne besondere Markierung zwischen beiden;
    • Zeile 36: ein expliziter Ausdruck [@( ... )]. Klammern sind immer dann erforderlich, wenn der Ausdruck Leerzeichen oder Operatoren enthält (hier der Bedingungsoperator [? :]);
    • Zeile 37: Um unformatierten Text innerhalb eines [C#]-Blocks zu schreiben, wird er in das spezielle Tag [<text>] eingeschlossen, das auf der Seite nicht erscheint;
    • Zeile 38: [@:] leistet dasselbe für den Rest der Zeile;
    • Zeilen 45–51: Eine leere Liste ist einfach ein [if]. Im [else]-Block, der sich bereits im [C#]-Modus befindet, wird [foreach] ohne [@] geschrieben;
    • Zeile 55: die Variablen des [@{ }]-Blocks und explizite Ausdrücke, einschließlich eines Methodenaufrufs: [ToString("0.00")] formatiert eine Zahl mit zwei Nachkommastellen;
    • Zeile 56: ein Pfad in Punktnotation [@Model.Address.Street] und ein Methodenaufruf [ToUpper()];
    • Zeile 57: [Razor] erkennt, dass ein [@] zwischen zwei alphanumerischen Zeichen kein Code ist: [contact@example.com] ist eine E-Mail-Adresse. Um ein [@] zu schreiben, das mehrdeutig wäre, verdoppeln wir es: [@@].

    Kommentieren wir diesen Code:

    • Zeile 9: Der [HtmlEncoder]-Service wird von [Razor] verwendet, um die Ausdrücke in [@] zu maskieren. Standardmäßig lässt er nur Zeichen des einfachen lateinischen Alphabets unverändert passieren: Deshalb wird [é] als [&#xE9;] kodiert (und [€] als [&#x20AC;]). Wir ersetzen diesen Service durch einen Encoder, der alle Zeichen durchlässt ([UnicodeRanges.All]). Die gefährlichen Zeichen ([<], [>], [&], ["], [']) bleiben maskiert: Der Schutz vor HTML-Injection bleibt vollständig erhalten. Alle folgenden Beispiele sowie die Fallstudie enthalten diese Zeile;
    • Zeile 14: eine Middleware-Komponente (siehe Kapitel 7), die Dateien aus dem Ordner [wwwroot] des Projekts ausliefert: Die Datei [wwwroot/css/styles.css] wird unter der URL [/css/styles.css] ausgeliefert. Existiert die Datei nicht, wird die Anfrage an die Controller weitergereicht;
    • Zeilen 16–19: Ein zweiter Service für statische Dateien. Das CSS-[Bootstrap]-Framework ist allen Beispielen gemeinsam: Es wird nur einmal im Ordner [examples/common/bootstrap] abgelegt und unter dem Präfix [/bootstrap] ausgeliefert. [PhysicalFileProvider] bezeichnet einen Ordner auf der Festplatte; [ContentRootPath] ist der Projektordner. So wird [examples/common/bootstrap/css/bootstrap.min.css] unter der URL [/bootstrap/css/bootstrap.min.css] ausgeliefert. Die Fallstudie hingegen legt [Bootstrap] in ihrem eigenen Ordner ab, [wwwroot].

    using System.Globalization;
using System.Text.Encodings.Web;
using Microsoft.AspNetCore.Html;

namespace Example16.Helpers;

// gewöhnliche C#-Funktionen, aus den Views aufrufbar: @Formats.Money(...)
public static class Formats
{
    // "keyboard" -> "KEYBOARD"
    public static string Uppercase(string? text) => (text ?? "").ToUpperInvariant();

    // ein Betrag in einer Währung, im Format der aktuellen Sprache (CultureInfo.CurrentCulture)
    // "C" = Währung; das Objekt NumberFormatInfo legt das Währungssymbol fest
    public static string Money(decimal amount, string currency = "EUR")
    {
        var format = (NumberFormatInfo)CultureInfo.CurrentCulture.NumberFormat.Clone();
        format.CurrencySymbol = currency == "USD" ? "$" : "€";
        return amount.ToString("C", format);
    }

    // ein ausgeschriebenes Datum in der aktuellen Sprache: "Wednesday, September 23, 2026"
    public static string Date(DateTime when) => when.ToString("D", CultureInfo.CurrentCulture);

    // ein Helper, der HTML erzeugt: er gibt ein IHtmlContent zurück, das Razor nicht maskiert.
    // Der Helper muss dann selbst die eingefügten Daten maskieren (HtmlEncoder)
    public static IHtmlContent Badge(int stock)
    {
        var cssClass = stock > 0 ? "text-bg-success" : "text-bg-danger";
        var text = HtmlEncoder.Default.Encode(stock > 0 ? $"{stock} in stock" : "sold out");
        return new HtmlString($"<span class=\"badge {cssClass}\">{text}</span>");
    }
}

// ein Tag Helper: er wird auf die <a>-Tags angewendet, die das Attribut "active-path" tragen
// <a href="/products" active-path="/products"> erhält die Klasse "active", wenn die
// aktuelle Seite unter diesem Pfad liegt
[HtmlTargetElement("a", Attributes = "active-path")]
public class ActiveTagHelper(IHttpContextAccessor accessor) : TagHelper
{
    // der Wert des Attributs active-path="..."
    public string ActivePath { get; set; } = "";

    public override void Process(TagHelperContext context, TagHelperOutput output)
    {
        var path = accessor.HttpContext?.Request.Path.Value ?? "/";
        var isActive = ActivePath == "/" ? path == "/" : path.StartsWith(ActivePath, StringComparison.Ordinal);
        if (isActive)
        {
            output.AddClass("active", HtmlEncoder.Default); // fügt die Klasse zum Attribut class hinzu
        }
    }
}

    4.7.1. Validierung durch den Browser

    Die Attribute [required], [maxlength], [min], [max], [type="email"] ... weisen den Browser an, das Formular vor dem Absenden zu validieren: Der Benutzer wird sofort über einen Tooltip benachrichtigt, ohne jeglichen Datenaustausch mit dem Server. Dieses Verhalten erreicht man durch einen Klick auf den Link „enable it“ (http://localhost:5000/register?html5=true), der das Attribut [novalidate] aus dem Formular entfernt.

    Diese Validierung ist eine Komfortfunktion, keine Sicherheitsmaßnahme: Man kann sie umgehen, indem man einfach das Attribut [required] mit den Entwicklertools des Browsers entfernt oder die Anfrage mit [curl] sendet. Die serverseitige Validierung bleibt unverzichtbar. Deshalb deaktiviert das Beispiel die Browser-Validierung standardmäßig: um die Funktionsweise des Servers zu demonstrieren. Die Fallstudie behält dagegen beide bei: das HTML5-Attribut aus Komfortgründen und die serverseitige Validierung für die Sicherheit.

    4.8. Beispiel 20: Das Post/Redirect/Get-Muster

    Wenn ein POST-Formular erfolgreich gesendet wurde, kann die Action direkt eine Seite anzeigen. Das Problem entsteht, wenn der Benutzer diese Seite anschließend aktualisiert (Taste F5): Der Browser sendet das Formular erneut, und die Operation wird ein zweites Mal ausgeführt (eine doppelt aufgegebene Bestellung, ein doppelt vereinbarter Termin ...). Das Muster Post / Redirect / Get (PRG) verhindert dieses Problem: Eine erfolgreiche POST-Action erzeugt keine Seite, sondern leitet auf eine GET-Seite um. Das Neuladen dieser Seite löst lediglich einen weiteren GET aus, ohne Auswirkung.

    Ein Detail bleibt: Die GET-Seite muss eine Meldung anzeigen („Die Aufgabe wurde hinzugefügt“), obwohl diese Meldung von der POST-Action in einer anderen Anfrage erzeugt wurde. Hierfür [ASP.NET Core] stellt [TempData] bereit: ein Dictionary, dessen Inhalt bis zur nächsten Anfrage erhalten bleibt und verschwindet, sobald er gelesen wurde. Es handelt sich um eine Flash-Meldung. Standardmäßig wird [TempData] in einem Cookie gespeichert, das vom Server verschlüsselt wird.

    {
        var index = contacts.FindIndex(c => c.Id == id);
        if (index < 0) return null;
        // ein Record ist unveränderlich: wir ersetzen ihn durch eine geänderte Kopie (mit with)
        contacts[index] = contacts[index] with { Name = name, Phone = phone };
        return contacts[index];
    }

    {
        var contact = new Contact(++lastId, name, phone);
        contacts.Add(contact);
        return contact;
    }

    public Contact? Delete(int id)
    {
        var contact = Read(id);
        if (contact is not null) contacts.Remove(contact);
        return contact;
    }

    // Bestätigung der Formulare, die das Attribut data-confirmation="..." tragen
// Ohne JavaScript werden diese Formulare einfach ohne Bestätigung gesendet.
(() => {
  const modal = document.getElementById('confirmation-modal');
  const message = document.getElementById('confirmation-modal-message');
  let pending = null; // das Formular, das bei Bestätigung durch den Benutzer gesendet wird

  // 1. wir fangen das Absenden der zu bestätigenden Formulare ab
  document.querySelectorAll('form[data-confirmation]').forEach((form) => {
    form.addEventListener('submit', (event) => {
      event.preventDefault(); // das Absenden wird ausgesetzt
      pending = form;
      message.textContent = form.dataset.confirmation;
      modal.classList.remove('d-none'); // das Fenster erscheint
    });
  });

  // 2. die beiden Buttons des Fensters
  modal.addEventListener('click', (event) => {
    const action = event.target.dataset.action;
    if (action === 'yes') {
      pending.submit(); // Absenden, OHNE das Ereignis 'submit' erneut auszulösen
    } else if (action === 'no') {
      modal.classList.add('d-none');
      pending = null;
    }
  });
})();
    • Zeile 3: Der Code ist in eine Funktion eingeschlossen, die sofort ausgeführt wird, um keine globalen Variablen zu erzeugen;
    • Zeilen 9–14: Das Absenden jedes Formulars mit dem Attribut [data-confirmation] wird abgefangen. [preventDefault()] setzt das Absenden aus; das Formular wird gespeichert, die Frage wird im Fenster platziert ([textContent], nicht [innerHTML]: Die Frage wird als Text angezeigt und nie als HTML interpretiert), und das Fenster wird angezeigt;
    • Zeilen 19–25: „Ja“ sendet das zwischengespeicherte Formular ab. Die Methode [submit()] eines Formulars sendet es ab, ohne das Ereignis [submit] auszulösen: Es erscheint keine erneute Bestätigungsabfrage. „Nein“ verbirgt das Fenster.

    Ist [JavaScript] deaktiviert, werden die Formulare einfach ohne Bestätigung gesendet: Die Anwendung funktioniert weiterhin. Das ist das Prinzip der schrittweisen Verbesserung: [JavaScript] bietet zusätzlichen Komfort, ist aber nicht unerlässlich. Die Fallstudie folgt diesem Prinzip (ihr Skript [wwwroot/js/app.js] ergänzt die Tastaturbedienung: die Esc-Taste und den Fokus).
