    8. Authentifizierung und Autorisierung

    Bei jeder Anfrage stellen sich zwei getrennte Fragen:

    • Wer ist der Benutzer? Das ist die Authentifizierung: Der Benutzer hat sich mit Benutzername und Passwort ausgewiesen, und jede folgende Anfrage muss nachweisen, dass sie von ihm stammt;
    • Hat der Benutzer die Berechtigung, auf diese Seite zuzugreifen? Das ist die Autorisierung: In der Fallstudie verwaltet nur ein Benutzer mit der Rolle ADMIN die Ärzte und Kunden.

    Da der Server zustandslos ist (Kapitel 6), muss der Identitätsnachweis in jeder Anfrage enthalten sein. Verwendet wird ein JWT-Token ([JSON Web Token]): Nach der Anmeldung erzeugt der Server ein Token, das die Identität des Benutzers enthält, und signiert es mit einem geheimen Schlüssel. Das Token wird in einem Cookie gespeichert; der Browser sendet es mit jeder Anfrage zurück; der Server prüft die Signatur. Der Benutzer kann sein Token lesen, aber nicht ändern: Jede Änderung macht die Signatur ungültig.

    [ASP.NET Core] unterscheidet die beiden Fragen klar:

    • Die Authentifizierung wird von einer Middleware ([UseAuthentication]) und einem Authentifizierungsschema übernommen: hier das Schema [JwtBearer], bereitgestellt vom [NuGet]-Paket [Microsoft.AspNetCore.Authentication.JwtBearer] . Bei jeder Anfrage sucht es nach einem Token, prüft es und leitet daraus den Benutzer ab, [HttpContext.User];
    • Die Autorisierung wird vom Attribut [[Authorize]] übernommen, das von der Middleware [UseAuthorization] geprüft wird: Sie vergleicht die Anforderungen der Action mit dem, was über den Benutzer bekannt ist.

    Die Beispiele dieses Kapitels verwenden zwei [NuGet]-Pakete: [Microsoft.AspNetCore.Authentication.JwtBearer] und [BCrypt.Net-Next] (Passwort-Hashing). Sie sind in der Projektdatei deklariert; der Befehl [dotnet build] (oder [dotnet run]) lädt sie bei der ersten Kompilierung von nuget.org herunter: Eine Internetverbindung ist daher erforderlich.

    namespace Example25.Auth;

public record User(string Login, string Name, string Role, string PasswordHash);

// die Benutzer im Speicher (das nächste Kapitel legt sie in einer Datenbank ab)
public class UsersService
{
    private readonly List<User> users =
    [
        // ein Passwort wird NIE gespeichert, nur sein Hash
        Create("marie", "Marie Pélissier", "USER", "marie"),
        Create("admin", "Administrator", "ADMIN", "admin"),
    ];

    public User? Find(string login) =>
        users.Find(u => u.Login == login);

    // bcrypt mit Kostenfaktor 10: 2^10 Berechnungsrunden, bewusst langsam
    private static User Create(string login, string name, string role, string password) =>
        new(login, name, role, BCrypt.Net.BCrypt.HashPassword(password, 10));
}
    • Zeile 3: ein Benutzer mit seiner Rolle;
    • Zeilen 10–12: Ein Passwort wird nie gespeichert; nur sein Hash wird behalten – ein Fingerabdruck, der von einer Einwegfunktion berechnet wird. Man kann prüfen, ob ein Passwort zu einem Fingerabdruck passt, aber das Passwort nicht aus dem Fingerabdruck wiederherstellen;
    • Zeilen 19–20: [BCrypt.Net.BCrypt.HashPassword(password, 10)] berechnet den Hash mit dem Algorithmus [bcrypt]. Die 10 steht für die „Kosten“: [bcrypt] ist absichtlich langsam (2¹⁰ Berechnungsrunden), um davon abzuschrecken, Millionen von Passwörtern auszuprobieren. Zwei Hashes desselben Passworts sind verschieden (es wird ein zufälliges „Salt“ hinzugefügt): [BCrypt.Verify] übernimmt deren Überprüfung.

    using System.Text;
using Microsoft.IdentityModel.JsonWebTokens;
using Microsoft.IdentityModel.Tokens;

namespace Example25.Auth;

public class AuthService(UsersService users, IConfiguration configuration)
{
    // der Name des Cookies, das das Token trägt
    public const string Cookie = "token";

    // der Schlüssel, der die Tokens signiert, aus appsettings.json gelesen (Abschnitt Jwt)
    public static SymmetricSecurityKey Key(IConfiguration configuration) =>
        new(Encoding.UTF8.GetBytes(configuration["Jwt:Secret"]!));

    // null, wenn der Login unbekannt oder das Passwort falsch ist
    public User? Validate(string? login, string? password)
    {
        var user = users.Find(login ?? "");
        if (user is null || password is null)
        {
            return null;
        }
        return BCrypt.Net.BCrypt.Verify(password, user.PasswordHash) ? user : null;
    }

    // das JWT-Token des Benutzers, mit dem geheimen Schlüssel signiert
    public string CreateToken(User user)
    {
        var descriptor = new SecurityTokenDescriptor
        {
            // der Inhalt des Tokens: die "Claims" (Aussagen über den Benutzer)
            Claims = new Dictionary<string, object>
            {
                ["sub"] = user.Login,
                ["name"] = user.Name,
                ["role"] = user.Role,
            },
            Expires = DateTime.UtcNow.AddMinutes(configuration.GetValue<int>("Jwt:DurationMinutes")),
            SigningCredentials = new SigningCredentials(Key(configuration), SecurityAlgorithms.HmacSha256),
        };
        return new JsonWebTokenHandler().CreateToken(descriptor);
    }
}

    using System.Text.Encodings.Web;
using System.Text.Unicode;
using Example25.Auth;
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.Extensions.FileProviders;
using Microsoft.IdentityModel.Tokens;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddControllersWithViews();
builder.Services.AddSingleton(HtmlEncoder.Create(UnicodeRanges.All));
builder.Services.AddSingleton<UsersService>();
builder.Services.AddSingleton<AuthService>();

// Authentifizierung: ein JWT-Token, bei jeder Anfrage überprüft
builder.Services
    .AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(options =>
    {
        options.MapInboundClaims = false; // die Claim-Namen unverändert lassen ("sub", "role"...)
        options.TokenValidationParameters = new TokenValidationParameters
        {
            ValidateIssuerSigningKey = true,
            IssuerSigningKey = AuthService.Key(builder.Configuration), // die Signatur muss von uns stammen
            ValidateLifetime = true,                                   // das Token darf nicht abgelaufen sein
            ValidateIssuer = false,
            ValidateAudience = false,
            NameClaimType = "name", // User.Identity.Name
            RoleClaimType = "role", // [Authorize(Roles = ...)], User.IsInRole(...)
        };
        options.Events = new JwtBearerEvents
        {
            // 1. das Token wird aus dem Cookie gelesen (und nicht aus dem Header Authorization)
            OnMessageReceived = context =>
            {
                context.Token = context.Request.Cookies[AuthService.Cookie];
                return Task.CompletedTask;
            },
            // 2. geschützte Seite ohne gültiges Token: weiter zur Login-Seite
            OnChallenge = context =>
            {
                context.HandleResponse(); // wir erstellen die Antwort selbst (kein 401)
                context.Response.Redirect("/login");
                return Task.CompletedTask;
            },
        };
    });
builder.Services.AddAuthorization();

var app = builder.Build();

app.UseStaticFiles(new StaticFileOptions
{
    FileProvider = new PhysicalFileProvider(Path.GetFullPath(Path.Combine(builder.Environment.ContentRootPath, "..", "common", "bootstrap"))),
    RequestPath = "/bootstrap",
});
app.UseAuthentication(); // wer ist es? -> HttpContext.User
app.UseAuthorization();  // ist es erlaubt? -> [Authorize]

app.MapControllers();
app.Run();

    Kommentieren wir diesen Code:

    • Zeilen 16–17: Wir deklarieren das Authentifizierungsschema [JwtBearer] und legen es als Standardschema fest;
    • Zeile 19: Standardmäßig benennt [JwtBearer] bestimmte Claims mit langen, von Microsoft-Technologien geerbten Namen um ([role] würde zu [http://schemas.microsoft.com/ws/2008/06/identity/claims/role]). Wir behalten die Namen des Tokens bei;
    • Zeilen 20–26: Die Prüfungen, die bei jedem Token durchgeführt werden: die Signatur (mit unserem Schlüssel) und das Ablaufdatum. Unsere Tokens geben weder den Aussteller ([issuer]) noch den Empfänger ([audience]) an, daher prüfen wir sie nicht;
    • Zeilen 27–28: welcher Claim den Namen des Benutzers enthält ([User.Identity.Name]) und welcher seine Rolle ([[Authorize(Roles = ...)]], [User.IsInRole(...)]);
    • Zeilen 33–36: die Ereignisse von [JwtBearer] ermöglichen es, das Verhalten anzupassen. Standardmäßig sucht es das Token im [Authorization: Bearer ...]-Header der Anfrage: So arbeitet ein [JavaScript]-Client oder eine mobile App. Ein Browser fügt diesen Header nicht von selbst hinzu: Das Token wird aus dem Cookie gelesen;
    • Zeilen 39–42: die Challenge tritt auf, wenn eine geschützte Seite ohne gültiges Token angefordert wird. Standardmäßig lautet die Antwort ein 401-Fehler ohne Inhalt: nützlich für einen [JavaScript]-Client, aber verwirrend für einen Benutzer. Der Benutzer wird daher auf die Anmeldeseite umgeleitet. [HandleResponse()] teilt [JwtBearer] mit, dass die Antwort bereits behandelt wird;
    • Zeilen 56–57: die beiden Middleware-Komponenten in dieser Reihenfolge: zuerst wird bestimmt, wer der Benutzer ist, dann werden seine Berechtigungen geprüft.

    using Example25.Auth;
using Microsoft.AspNetCore.Mvc;

namespace Example25.Controllers;

public class AuthController(AuthService auth, IConfiguration configuration) : Controller
{
    [HttpGet("/")]
    public IActionResult Home() => Redirect("/profile");

    [HttpGet("login")]
    public IActionResult Form() => View("Login");

    [HttpPost("login")]
    public IActionResult Login(string? login, string? password)
    {
        var user = auth.Validate(login, password);
        if (user is null)
        {
            // eine bewusst vage Meldung: wir sagen nicht, ob es der Login oder das Passwort ist
            ViewData["Error"] = "Incorrect login or password.";
            ViewData["Login"] = login;
            return View("Login");
        }
        // das Token in einem Cookie, das das JavaScript der Seite nicht lesen kann
        Response.Cookies.Append(AuthService.Cookie, auth.CreateToken(user), new CookieOptions
        {
            HttpOnly = true,
            SameSite = SameSiteMode.Lax,
            MaxAge = TimeSpan.FromMinutes(configuration.GetValue<int>("Jwt:DurationMinutes")),
        });
        return Redirect("/profile");
    }

    // das Abmelden ändert den Zustand des Benutzers: es ist ein POST
    [HttpPost("logout")]
    public IActionResult Logout()
    {
        Response.Cookies.Delete(AuthService.Cookie);
        return Redirect("/login");
    }
}

    • Zeilen 36–40: Die Abmeldung löscht das Cookie ([Cookies.Delete]). Dies ist ein POST: Er ändert den Zustand des Benutzers (siehe Beispiel 21).

    8.3. Beispiel 27: CAPTCHA und Begrenzung der Versuche

    Eine Anmeldeseite zieht Bots an: Programme, die Tausende von Login-/Passwort-Kombinationen ausprobieren. Zwei sich ergänzende Schutzmaßnahmen:

    • ein CAPTCHA: ein Bild mit verzerrten Zeichen, das ein Mensch leicht lesen kann, ein einfacher Bot jedoch nicht entziffern kann;
    • eine Begrenzung der Versuche: Nach einigen fehlgeschlagenen Versuchen wird die IP-Adresse für eine Weile gesperrt.

    using System.Collections.Concurrent;

namespace Example27.Auth;

// ein CAPTCHA: seine Kennung und sein Bild (SVG)
public record Captcha(string Id, string Svg);

// Singleton: die ausstehenden CAPTCHAs werden vom Server im Speicher gehalten
public class CaptchaService
{
    private static readonly TimeSpan Lifetime = TimeSpan.FromMinutes(10);

    // Kennung -> (erwartete Antwort, Ablaufdatum).
    // ConcurrentDictionary: ein Dictionary, das von mehreren gleichzeitigen Anfragen genutzt werden kann
    private readonly ConcurrentDictionary<string, (string Text, DateTime Expires)> pending = new();

    public Captcha Generate()
    {
        var text = CaptchaDrawing.RandomText(5);
        var id = Guid.NewGuid().ToString(); // eine zufällige, nicht zu erratende Kennung
        pending[id] = (text, DateTime.UtcNow + Lifetime);
        return new Captcha(id, CaptchaDrawing.Svg(text));
    }

    // das CAPTCHA wird in jedem Fall VERBRAUCHT: ein Bild wird nur einmal verwendet
    public bool Check(string? id, string? input)
    {
        if (id is null || input is null || !pending.TryRemove(id, out var expected))
        {
            return false;
        }
        return expected.Expires > DateTime.UtcNow
            && string.Equals(expected.Text, input.Trim(), StringComparison.OrdinalIgnoreCase);
    }
}

    Gehen wir diesen Code Schritt für Schritt durch:

    • Zeile 15: Die erwarteten Antworten werden vom Server unter einer zufälligen Kennung gespeichert. Die Seite enthält nur das Bild und die Kennung: Die Antwort verlässt den Server nie. Der Service ist ein Singleton, das von gleichzeitigen Anfragen verwendet wird: [ConcurrentDictionary] ist ein Dictionary, das genau dafür gedacht ist;
    • Zeilen 17–22: Eine Zeichenkette aus fünf Zeichen wird zufällig ausgewählt, gespeichert und als Bild gezeichnet. Das Zeichnen übernimmt die Klasse [CaptchaDrawing] (Datei [Auth/CaptchaDrawing.cs], aus der Fallstudie übernommen und dort Zeile für Zeile kommentiert): Jedes Zeichen besteht aus einer Menge von Linien, die vergrößert, geneigt und zufällig verzerrt werden, und das Bild wird durch drei Kurven verwischt. Leicht verwechselbare Zeichen (0 und O, 1, I und L) sind ausgeschlossen;
    • Zeilen 26–33: Die Überprüfung verbraucht das CAPTCHA ([TryRemove] entfernt es aus dem Dictionary): Ein Bild wird nur einmal verwendet; ein Bot kann eine bereits gefundene Lösung nicht wiederverwenden. Die Groß- und Kleinschreibung spielt keine Rolle.

    namespace Example27.Auth;

// Singleton: zählt die Login-Fehlversuche jeder IP-Adresse
public class LimiterService
{
    private const int MaxFailures = 3;
    private static readonly TimeSpan BlockDuration = TimeSpan.FromMinutes(1);

    private class Counter
    {
        public int Failures;
        public DateTime BlockedUntil;
    }

    private readonly Dictionary<string, Counter> counters = [];
    private readonly Lock padlock = new(); // mehrere gleichzeitige Anfragen: ein Zugriff nach dem anderen

    // die Anzahl der verbleibenden Sperrsekunden (0: nicht gesperrt)
    public int BlockedSeconds(string ip)
    {
        lock (padlock)
        {
            if (!counters.TryGetValue(ip, out var c)) return 0;
            var remaining = c.BlockedUntil - DateTime.UtcNow;
            return remaining > TimeSpan.Zero ? (int)Math.Ceiling(remaining.TotalSeconds) : 0;
        }
    }

    public void Failure(string ip)
    {
        lock (padlock)
        {
            if (!counters.TryGetValue(ip, out var c))
            {
                c = new Counter();
                counters[ip] = c;
            }
            c.Failures++;
            if (c.Failures >= MaxFailures)
            {
                c.BlockedUntil = DateTime.UtcNow + BlockDuration;
                c.Failures = 0; // nach der Sperre beginnen wir wieder bei null
            }
        }
    }

    public void Success(string ip)
    {
        lock (padlock)
        {
            counters.Remove(ip);
        }
    }
}
    • Zeilen 6–7: Drei fehlgeschlagene Versuche sperren die IP-Adresse für eine Minute;
    • Zeile 16: Das Dictionary der Zähler wird von allen Anfragen gemeinsam genutzt. Die Anweisung [lock (padlock) { ... }] stellt sicher, dass immer nur eine Anfrage den Block ausführt: Zwei gleichzeitige Fehlversuche können so nicht verloren gehen (siehe [Interlocked], Kapitel 6). [Lock] ist der Sperrtyp ab [.NET] 9 und in späteren Versionen;
    • Zeilen 19–25: die verbleibende Sperrzeit;
    • Zeilen 29–42: Ein Fehlversuch wird gezählt; beim dritten Fehlversuch wird die Adresse gesperrt;
    • Zeilen 47–51: Ein Erfolg setzt den Zähler auf null zurück.

    using Example27.Auth;
using Microsoft.AspNetCore.Mvc;

namespace Example27.Controllers;

public class AuthController(AuthService auth, CaptchaService captchas, LimiterService limiter,
    IConfiguration configuration) : Controller
{
    [HttpGet("/")]
    public IActionResult Home() => Redirect("/profile");

    [HttpGet("login")]
    public IActionResult Form() => LoginPage(null, null);

    [HttpPost("login")]
    public IActionResult Login(string? login, string? password, string? captchaId, string? captcha)
    {
        // die IP-Adresse des Clients
        var ip = HttpContext.Connection.RemoteIpAddress?.ToString() ?? "?";

        // 1. eine gesperrte Adresse wird gar nicht erst beachtet
        var seconds = limiter.BlockedSeconds(ip);
        if (seconds > 0)
        {
            return LoginPage(login, $"Too many attempts: try again in {seconds} s.");
        }
        // 2. zuerst das CAPTCHA (billig), dann das Passwort (bcrypt, teuer)
        if (!captchas.Check(captchaId, captcha))
        {
            limiter.Failure(ip);
            return LoginPage(login, "The image code is incorrect.");
        }
        var user = auth.Validate(login, password);
        if (user is null)
        {
            limiter.Failure(ip);
            return LoginPage(login, "Incorrect login or password.");
        }
        limiter.Success(ip);
        Response.Cookies.Append(AuthService.Cookie, auth.CreateToken(user), new CookieOptions
        {
            HttpOnly = true,
            SameSite = SameSiteMode.Lax,
            MaxAge = TimeSpan.FromMinutes(configuration.GetValue<int>("Jwt:DurationMinutes")),
        });
        return Redirect("/profile");
    }

    [HttpPost("logout")]
    public IActionResult Logout()
    {
        Response.Cookies.Delete(AuthService.Cookie);
        return Redirect("/login");
    }

    // jede Anzeige des Formulars erzeugt ein NEUES CAPTCHA
    private ViewResult LoginPage(string? login, string? error)
    {
        ViewData["Login"] = login;
        ViewData["Error"] = error;
        return View("Login", captchas.Generate());
    }
}
