    9. Schichtenarchitektur und Datenzugriff

    Die bisherigen Beispiele hielten ihre Daten im Speicher. Eine reale Anwendung speichert sie in einer Datenbank und ist in Schichten organisiert (siehe das Diagramm in der Einführung):

    • die Web-Schicht (Controller, Views): liest die Anfrage, ruft die Geschäftsschicht auf, wählt die View aus;
    • die Geschäfts-Schicht: Geschäftsregeln;
    • die DAO-Schicht ([Data Access Object]): Datenbankabfragen.

    Jede Schicht kennt nur die darunterliegende Schicht und möglichst wenige Details über sie: Die Web-Schicht kennt kein SQL, und die Geschäftsschicht kennt kein HTML. Diese Trennung macht jede Schicht leichter verständlich, testbar und weiterentwickelbar. Beispiel 28 wendet dies auf die Verwaltung einer Personenliste an: Es ist, im kleineren Maßstab, die Verwaltung der Ärzte und Klienten aus der Fallstudie.

    Der Datenbankzugriff erfolgt über Entity Framework Core (EF Core), den ORM ([Object-Relational Mapper]) von [.NET]: Er bildet [C#]-Klassen (die Entitäten) auf die Datenbanktabellen ab, übersetzt in [LINQ] geschriebene Abfragen in SQL und wandelt die abgerufenen Zeilen in Objekte um. EF Core kommuniziert mit jedem Datenbanktyp über einen Provider ([provider]): für [MySQL] das [NuGet]-Paket [Pomelo.EntityFrameworkCore.MySql], das wiederum EF Core und den Treiber [MySqlConnector] einbindet.

    using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;

namespace Example28.Entities;

// eine Zeile der Tabelle [persons]
[Table("persons")]
public class Person
{
    // der Primärschlüssel: ein Integer-Schlüssel wird konventionsgemäß von der Datenbank erzeugt
    // (AUTO_INCREMENT); EF Core liest seinen Wert nach jedem INSERT zurück
    [Key]
    [Column("id")]
    public int Id { get; set; }

    // das optimistische Sperren: jedes UPDATE trägt die Bedingung "AND version = <gelesener Wert>"
    [Column("version")]
    [ConcurrencyCheck]
    public int Version { get; set; } = 1;

    [Column("first_name")]
    [MaxLength(30)]
    public string FirstName { get; set; } = "";

    [Column("last_name")]
    [MaxLength(30)]
    public string LastName { get; set; } = "";

    [Column("age")]
    public int Age { get; set; }
}
    • Zeile 7: Die Klasse [Person] entspricht der Tabelle [persons]: Jede Zeile der Tabelle wird zu einem [Person]-Objekt;
    • Zeilen 12–14: der Primärschlüssel ([[Key]]). [[Column]] gibt den Spaltennamen an, der jeder Eigenschaft entspricht;
    • Zeilen 18–19: die Versionsspalte. [[ConcurrencyCheck]] weist EF Core an, bei jedem [AND version = <value read>] die Bedingung [UPDATE] hinzuzufügen: Das ist das optimistische Sperren, das weiter unten erläutert wird;
    • Zeile 22: die übrigen Spalten. [[MaxLength]] gibt die Spaltengröße an.

    using Example28.Entities;
using Microsoft.EntityFrameworkCore;

namespace Example28.Data;

// der DbContext: die Verbindung zur Datenbank und die Nachverfolgung der gelesenen Entitäten
public class PersonsDbContext(DbContextOptions<PersonsDbContext> options) : DbContext(options)
{
    // die "Tabelle" der Personen: sie wird mit LINQ abgefragt
    public DbSet<Person> Persons => Set<Person>();

    // unmittelbar vor jedem Speichern: die Version jeder GEÄNDERTEN Person
    // wird um 1 erhöht. Das erzeugte UPDATE vergleicht den ALTEN Wert:
    //   UPDATE persons SET ..., version = 2 WHERE id = 1 AND version = 1
    public override Task<int> SaveChangesAsync(bool acceptAllChangesOnSuccess, CancellationToken cancellationToken = default)
    {
        foreach (var entry in ChangeTracker.Entries<Person>())
        {
            if (entry.State == EntityState.Modified)
            {
                entry.Entity.Version++;
            }
        }
        return base.SaveChangesAsync(acceptAllChangesOnSuccess, cancellationToken);
    }
}
    • Zeile 7: DbContext ist das zentrale Objekt von EF Core. Er stellt eine Arbeitssitzung mit der Datenbank dar: Er öffnet Verbindungen, sendet Abfragen und verfolgt die gelesenen Entitäten (er weiß, welche geändert wurden);
    • Zeile 10: Ein [DbSet<Person>] stellt die Tabelle dar: Sie wird mit [LINQ] abgefragt, und Entitäten werden ihr hinzugefügt;
    • Zeilen 15–24: [SaveChangesAsync] sendet alle ausstehenden Änderungen an die Datenbank. Die Methode ist so überschrieben, dass sie die Versionsnummer jeder geänderten Person erhöht. [ChangeTracker] verfolgt die Änderungen: Für jede verfolgte Entität kennt er ihren Status ([Added], [Modified], [Deleted], [Unchanged]). EF Core verwaltet, anders als [TypeORM], nicht von sich aus eine ganzzahlige Versionsspalte: Wir müssen sie selbst erhöhen.

    using Example28.Data;
using Example28.Entities;
using Microsoft.EntityFrameworkCore;

namespace Example28.Dao;

// die DAO-Schicht: Abfragen, keine Geschäftsregel
public class PersonsDao(PersonsDbContext db)
{
    // SELECT ... FROM persons ORDER BY last_name, first_name
    public Task<List<Person>> ListAsync() =>
        db.Persons.AsNoTracking().OrderBy(p => p.LastName).ThenBy(p => p.FirstName).ToListAsync();

    // SELECT ... FROM persons WHERE id = @id LIMIT 1; null, falls nicht vorhanden
    public Task<Person?> ReadAsync(int id) =>
        db.Persons.FirstOrDefaultAsync(p => p.Id == id);

    // eine Person mit gleichem Vor- und Nachnamen, außer [exceptId]
    public Task<Person?> NamesakeAsync(string firstName, string lastName, int exceptId) =>
        db.Persons.FirstOrDefaultAsync(p => p.FirstName == firstName && p.LastName == lastName && p.Id != exceptId);

    // INSERT, wenn die Person neu ist (Id = 0), sonst UPDATE der geänderten Spalten
    public async Task<Person> SaveAsync(Person person)
    {
        if (person.Id == 0)
        {
            db.Persons.Add(person);
        }
        await db.SaveChangesAsync();
        return person;
    }

    // DELETE FROM persons WHERE id = @id, ohne die Person vorher zu lesen
    public Task DeleteAsync(int id) =>
        db.Persons.Where(p => p.Id == id).ExecuteDeleteAsync();
}
    • Zeile 8: Das DAO erhält [DbContext] per Injektion;
    • Zeilen 11–12: Eine [LINQ]-Abfrage, die von EF Core in SQL übersetzt wird. Vor dem Aufruf von [ToListAsync], der die Abfrage ausführt und die Liste zurückgibt, wird nichts an die Datenbank gesendet. [AsNoTracking()] gibt an, dass diese Entitäten nicht verändert werden: [DbContext] muss sie nicht verfolgen;
    • Zeilen 15–16: das erste Element, das die Bedingung erfüllt, oder [null];
    • Zeilen 19–20: Eine Person mit demselben Vor- und Nachnamen, außer [exceptId]: nützlich beim Ändern eines Datensatzes, der nicht mit sich selbst kollidieren darf. Der Vergleich wird von [MySQL] durchgeführt, also gemäß der Kollation der Datenbank: ohne Rücksicht auf Groß- und Kleinschreibung und auf Akzente;
    • Zeilen 23–30: Eine neue Person ([Id] auf 0 gesetzt) wird zu [DbSet] hinzugefügt; [SaveChangesAsync] erzeugt dann ein [INSERT] und liest die von der Datenbank vergebene Kennung erneut ein. Eine von demselben [DbContext] gelesene und geänderte Person wird durch die Änderungsverfolgung erkannt: [SaveChangesAsync] erzeugt ein [UPDATE], das nur die geänderten Spalten enthält;
    • Zeilen 34–35: [ExecuteDeleteAsync] sendet ein [DELETE] direkt an die Datenbank, ohne die Datensätze zuvor zu lesen.

    Die DAO-Schicht enthält keine Regeln: nur Abfragen. Diese Methoden sind asynchron ([Task], [async] / [await]): Während die Datenbank arbeitet, wird der Thread freigegeben und bedient andere Anfragen.

    using Example28.Dao;
using Example28.Data;
using Example28.Entities;
using Microsoft.EntityFrameworkCore;
using MySqlConnector;

namespace Example28.Business;

public record PersonData(string FirstName, string LastName, int Age);

// die GESCHÄFTSSCHICHT: die Geschäftsregeln, und nur sie
public class PersonsBusiness(PersonsDao dao, PersonsDbContext db)
{
    private const string Namesake = "A person already has these first and last names.";

    public Task<List<Person>> ListAsync() => dao.ListAsync();

    public async Task<Person> ReadAsync(int id) =>
        await dao.ReadAsync(id) ?? throw new BusinessException("This person does not exist (anymore).");

    public Task<Person> AddAsync(PersonData data) =>
        TransactionAsync(async () =>
        {
            // Regel 1: keine zwei Personen mit gleichem Vor- und Nachnamen
            if (await dao.NamesakeAsync(data.FirstName, data.LastName, 0) is not null)
            {
                throw new BusinessException(Namesake);
            }
            var person = new Person { FirstName = data.FirstName, LastName = data.LastName, Age = data.Age };
            return await dao.SaveAsync(person);
        });

    public Task<Person> EditAsync(int id, int version, PersonData data) =>
        TransactionAsync(async () =>
        {
            var person = await ReadAsync(id);
            // Regel 2: wir ändern keinen Datensatz, der inzwischen von jemand anderem geändert wurde
            if (person.Version != version)
            {
                throw new BusinessException("This record has been modified meanwhile: reload it.");
            }
            if (await dao.NamesakeAsync(data.FirstName, data.LastName, id) is not null)
            {
                throw new BusinessException(Namesake);
            }
            person.FirstName = data.FirstName;
            person.LastName = data.LastName;
            person.Age = data.Age;
            return await dao.SaveAsync(person);
        });

    public async Task<Person> DeleteAsync(int id)
    {
        var person = await ReadAsync(id);
        await dao.DeleteAsync(id);
        return person;
    }

    // [work] läuft innerhalb einer Transaktion: COMMIT bei Erfolg, ROLLBACK
    // bei einer Ausnahme. Die Datenbankfehler werden zu BusinessExceptions
    private async Task<R> TransactionAsync<R>(Func<Task<R>> work)
    {
        await using var transaction = await db.Database.BeginTransactionAsync(); // START TRANSACTION
        try
        {
            var result = await work();
            await transaction.CommitAsync(); // COMMIT
            return result;
        }
        catch (DbUpdateConcurrencyException)
        {
            // das UPDATE "WHERE id = ... AND version = ..." hat keine Zeile geändert
            throw new BusinessException("This record has been modified meanwhile: reload it.");
        }
        catch (DbUpdateException e) when (e.InnerException is MySqlException { ErrorCode: MySqlErrorCode.DuplicateKeyEntry } sqlError
                                          && sqlError.Message.Contains("unq_persons_first_name_last_name"))
        {
            // zwei gleichzeitige Hinzufügungen: die Eindeutigkeitsbedingung der Datenbank hat die zweite abgewiesen
            throw new BusinessException(Namesake);
        }
        // jede andere Ausnahme nimmt ihren Lauf; die Transaktion, durch "await using"
        // ohne COMMIT geschlossen, wird abgebrochen (ROLLBACK)
    }
}

    Gehen wir diesen Code Schritt für Schritt durch:

    • Zeile 9: die Daten einer Person, wie sie von der Web-Schicht übergeben werden;
    • Zeile 12: Die Geschäftsschicht verwendet das DAO und [DbContext] , um Transaktionen zu öffnen. Beide sind [Scoped]-Services: Während einer Anfrage erhalten das DAO und die Geschäftsschicht den selben [DbContext] und damit dieselbe Verbindung und dieselbe Transaktion. (Mit [TypeORM] musste das Transaktionsobjekt an jede Methode des DAO übergeben werden);
    • Zeilen 18–19: Eine nicht vorhandene Person ist eine [BusinessException]. Der Operator [??] liefert einen Ersatzwert; [throw] kann als Ersatz dienen;
    • Zeilen 21–30: das Hinzufügen innerhalb einer Transaktion. Regel: Keine zwei Personen dürfen denselben Vor- und Nachnamen haben. Dies wird vor dem Schreiben geprüft, um eine klare Meldung anzuzeigen;
    • Zeilen 33–49: die Aktualisierung. Zusätzliche Regel: das optimistische Sperren (Zeilen 38–40). Das Änderungsformular hat die Version des Datensatzes so abgerufen, wie sie beim Öffnen des Formulars war. Hat sie sich in der Zwischenzeit geändert (ein anderer Benutzer hat denselben Datensatz geändert), wird die Aktualisierung abgelehnt, statt die Arbeit des anderen stillschweigend zu überschreiben;
    • Zeilen 61–67: [BeginTransactionAsync] öffnet eine Transaktion ([START TRANSACTION]), und [CommitAsync] bestätigt sie ([COMMIT]). [await using] schließt die Transaktion beim Verlassen des Blocks, was auch geschieht: Wird sie ohne [COMMIT] geschlossen, wird sie zurückgerollt ([ROLLBACK]). So macht eine von [work] ausgelöste Ausnahme (auch eine [BusinessException]) alles rückgängig, was bis dahin getan wurde;
    • Zeilen 70–73: Die zuvor erwähnte Versionsprüfung reicht nicht immer aus: Zwei Benutzer können denselben Datensatz gleichzeitig speichern, und beide bestehen die Prüfung. Das [UPDATE ... WHERE id = ... AND version = ...] des zweiten Benutzers ändert dann keine Zeile; EF Core erkennt dies und löst eine [DbUpdateConcurrencyException] aus, die zu derselben [BusinessException] wird;
    • Zeilen 75–79: Ebenso bestehen zwei gleichzeitige Hinzufügungen derselben Person die Namensprüfung, doch die Eindeutigkeitsbedingung der Datenbank lässt sie nicht zu. Der Fehler [MySQL] (Code [DuplicateKeyEntry], Nr. 1062) tritt verschachtelt in einer [DbUpdateException] auf (Eigenschaft [InnerException]); seine Meldung enthält den Namen der Bedingung. Die Klausel [when] fängt die Ausnahme nur ab, wenn die Bedingung zutrifft. [is MySqlException { ErrorCode: ... } sqlError] ist ein Pattern-Filter: Er prüft den Typ und den Wert einer Eigenschaft und speichert das Objekt in einer Variablen.

    using Example28.Business;
using Example28.Models;
using Microsoft.AspNetCore.Mvc;

namespace Example28.Controllers;

// die WEB-Schicht: sie kennt nur die Geschäftsschicht
public class PersonsController(PersonsBusiness business) : Controller
{
    [HttpGet("/")]
    public IActionResult Home() => RedirectToAction(nameof(List));

    [HttpGet("persons")]
    public async Task<IActionResult> List()
    {
        return View(await business.ListAsync());
    }

    [HttpGet("persons/add")]
    public IActionResult AddForm() => Form(BuildView("Add a person", "/persons/add", new()));

    [HttpPost("persons/add")]
    public async Task<IActionResult> Add(PersonForm input)
    {
        var view = BuildView("Add a person", "/persons/add", input);
        if (!ModelState.IsValid)
        {
            return Form(view);
        }
        try
        {
            var person = await business.AddAsync(new PersonData(input.FirstName!, input.LastName!, input.Age!.Value));
            TempData["Success"] = $"{person.FirstName} {person.LastName} has been added.";
            return RedirectToAction(nameof(List));
        }
        catch (BusinessException e)
        {
            // eine Geschäftsregel lehnt ab: wir zeigen das Formular erneut an, die Meldung unten auf der Seite
            return Form(view, e.Message);
        }
    }

    [HttpGet("persons/{id:int}/edit")]
    public async Task<IActionResult> EditForm(int id)
    {
        try
        {
            var p = await business.ReadAsync(id);
            var values = new PersonForm { FirstName = p.FirstName, LastName = p.LastName, Age = p.Age, Version = p.Version };
            return Form(BuildView("Edit a person", $"/persons/{id}/edit", values));
        }
        catch (BusinessException e)
        {
            TempData["Error"] = e.Message;
            return RedirectToAction(nameof(List));
        }
    }

    [HttpPost("persons/{id:int}/edit")]
    public async Task<IActionResult> Edit(int id, PersonForm input)
    {
        var view = BuildView("Edit a person", $"/persons/{id}/edit", input);
        if (!ModelState.IsValid)
        {
            return Form(view);
        }
        try
        {
            var data = new PersonData(input.FirstName!, input.LastName!, input.Age!.Value);
            var person = await business.EditAsync(id, input.Version ?? 0, data);
            TempData["Success"] = $"{person.FirstName} {person.LastName} has been edited.";
            return RedirectToAction(nameof(List));
        }
        catch (BusinessException e)
        {
            return Form(view, e.Message);
        }
    }

    [HttpPost("persons/{id:int}/delete")]
    public async Task<IActionResult> Delete(int id)
    {
        try
        {
            var person = await business.DeleteAsync(id);
            TempData["Success"] = $"{person.FirstName} {person.LastName} has been deleted.";
        }
        catch (BusinessException e)
        {
            // kein erneut anzuzeigendes Formular: der Fehler reist ebenfalls in TempData mit
            TempData["Error"] = e.Message;
        }
        return RedirectToAction(nameof(List));
    }

    // das Modell der Formular-View, mit den ModelState-Fehlern nach Feldern sortiert
    private FormView BuildView(string title, string action, PersonForm values) => new()
    {
        Title = title,
        Action = action,
        Values = values,
        Errors = ModelState
            .Where(c => c.Value!.Errors.Count > 0)
            .ToDictionary(c => c.Key, c => c.Value!.Errors[0].ErrorMessage, StringComparer.OrdinalIgnoreCase),
    };

    private ViewResult Form(FormView view, string? error = null)
    {
        ViewData["Error"] = error;
        return View("Form", view);
    }
}
